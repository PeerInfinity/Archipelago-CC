// scratch: run an authored tape on the GAME (p4f headless) and on the MODEL (defaults); print a trace.
// usage: SEEDLING_PORT=9687 node gamerun.mjs <tape.json> [--out=<file>] [--every=N]
import { createRequire } from 'node:module'; const { chromium } = createRequire('/home/user/Archipelago-CC/package.json')('playwright');
import { readFileSync, writeFileSync } from 'node:fs';
const REPO = '/home/user/Archipelago-CC';
const { HEADLESS_LOGIC_ONLY_ARGS } = await import(`${REPO}/scripts/procgen/headlessChromium.js`);
const { takeBoxLockOrExit } = await import(`${REPO}/scripts/procgen/boxLock.js`);
const { assertLogicOnlyChannel } = await import(`${REPO}/scripts/procgen/seedlingChannel.js`);
const M = `${REPO}/frontend/modules/seedlingDemo`;
const { gameVisibleTape, parseTape } = await import(`${M}/tapeFormat.js`);
const { createTapeStepper } = await import(`${M}/tapeRunner.js`);
const { atlasLevelSource } = await import(`${M}/levelSource.js`);

const file = process.argv[2];
const arg = (k) => process.argv.find((a) => a.startsWith(`--${k}=`))?.slice(k.length + 3) ?? null;
const OUT = arg('out');
const EVERY = +(arg('every') ?? 10);
const tape = parseTape(readFileSync(file, 'utf8'));

// MODEL (defaults)
const col = [];
let modelErr = null;
try {
    let run = null;
    const st = createTapeStepper(tape, { levelSource: atlasLevelSource(), onTick: (t, s, h, rn) => { run = rn; } });
    let r = st.next();
    while (!r.done) {
        const o = r.value.observation;
        col[o.t] = { x: o.x, y: o.y, level: run ? run.level : tape.boot.level, hits: run?.damage ? `${run.damage.hits}/${run.damage.hitsTimer}` : null,
            dt: (run?.darkTraps ?? []).map((b) => `${b.id}${b.removed ? ':gone' : b.dying ? ':die1' : b.startDying ? ':dying' : ''}`).join(' ') };
        r = st.next();
    }
} catch (e) { modelErr = String(e?.message ?? e).slice(0, 600); }

if (process.argv.includes('--model-only')) { for (let t = 0; t < col.length; t += EVERY) { const c = col[t]; if (c) console.log(`t${t} L${c.level} (${c.x.toFixed(2)},${c.y.toFixed(2)}) hits ${c.hits} dt[${c.dt}]`); } console.log('MODEL', modelErr ?? 'ok', col.length - 1); process.exit(0); }
takeBoxLockOrExit({ name: 'darktrapq-gamerun (scratch)', kind: 'browser' });
const PAGE_URL = `http://localhost:${process.env.SEEDLING_PORT || '8000'}/frontend/modules/flashPanel/wasm/seedling_bot_ap_p4f/game.html`;
const browser = await chromium.launch({ args: HEADLESS_LOGIC_ONLY_ARGS });
const frames = [];
let status = null;
try {
    const page = await browser.newPage();
    await page.goto(PAGE_URL, { waitUntil: 'domcontentloaded' });
    for (let i = 0; i < 480 && !(await page.evaluate(() => !!window.__runtimeReady)); i++) await page.waitForTimeout(250);
    await page.click('#btn-start');
    for (let i = 0; i < 480 && !(await page.evaluate(() => !!(window.__swfBridge?.game?.botStatus))); i++) await page.waitForTimeout(250);
    await assertLogicOnlyChannel(page);
    const bot = (n, a) => page.evaluate(([name, x]) => { const g = window.__swfBridge.game; return String(x === undefined ? g[name]() : g[name](x)); }, [n, a]);
    const loaded = await bot('botLoadTape', JSON.stringify(gameVisibleTape(tape)));
    if (loaded !== 'ok') throw new Error(`botLoadTape: ${loaded}`);
    const started = await bot('botStart');
    if (started !== 'ok') throw new Error(`botStart: ${started}`);
    const deadline = Date.now() + 300000;
    while (Date.now() < deadline) {
        const raw = await page.evaluate(() => { const g = window.__swfBridge.game; return [String(g.botMobiles()), String(g.botStatus())]; });
        const m = JSON.parse(raw[0]); const s = JSON.parse(raw[1]);
        frames.push({ s, m });
        if (s.finished) { status = s; break; }
    }
} finally { await browser.close(); }

const seen = new Set();
const rows = [];
for (const { s, m } of frames) {
    const t = m.tick ?? s.tick;
    if (seen.has(t)) continue; seen.add(t);
    const p = (m.mobiles ?? []).find((r) => /Player$/.test(r.cls));
    const dts = (m.mobiles ?? []).filter((r) => /DarkTrap$/.test(r.cls)).map((r) => `(${r.x},${r.y})${r.anim ? ':' + r.anim + r.anim_index : ''}`).join(' ');
    rows.push({ t, level: s.level, x: p?.x, y: p?.y, hits: s.hits, ri: s.receive_input, pc: JSON.stringify(s.persistence_cleared ?? []), dts, model: col[t] });
}
rows.sort((a, b) => a.t - b.t);
let first = null;
for (const r of rows) {
    if (!r.model || r.x === undefined) continue;
    const d = Math.hypot(r.model.x - r.x, r.model.y - r.y);
    if (d > 1e-9 && !first) first = { t: r.t, d, g: [r.x, r.y], m: [r.model.x, r.model.y] };
}
let last = -1e9;
for (const r of rows) {
    if (r.t - last < EVERY && r !== rows[rows.length - 1]) continue;
    last = r.t;
    console.log(`t${r.t} L${r.level} game(${r.x?.toFixed?.(2)},${r.y?.toFixed?.(2)}) hits ${r.hits} ri ${r.ri} dt[${r.dts}] | model ${r.model ? `L${r.model.level} (${r.model.x.toFixed(2)},${r.model.y.toFixed(2)}) dt[${r.model.dt}]` : '—'}`);
}
console.log('END status', JSON.stringify({ tick: status?.tick, level: status?.level, hits: status?.hits, x: status?.x, y: status?.y, pc: status?.persistence_cleared }));
console.log('MODEL', modelErr ? `ERR ${modelErr}` : `ticks ${col.length - 1}, end L${col.at(-1)?.level} (${col.at(-1)?.x?.toFixed(2)},${col.at(-1)?.y?.toFixed(2)})`);
console.log('FIRST DIVERGENCE', first ? JSON.stringify(first) : `none over ${rows.length} sampled ticks (0 px)`);
console.log('status keys', Object.keys(status ?? {}).join(' '));
if (OUT) writeFileSync(OUT, JSON.stringify({ tape: JSON.parse(readFileSync(file, 'utf8')), rows, status, first, modelErr }, null, 1));

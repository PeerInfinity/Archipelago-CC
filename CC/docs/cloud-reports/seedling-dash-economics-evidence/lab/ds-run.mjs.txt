#!/usr/bin/env node
// DS lab runner (scratch, never committed as code). One captured request, one window, in THIS process.
// node ds-run.mjs --lab=<labdir> --requests=<jsonl> --one=<id>:<i> --window=<n|0> [--budget=640] [--detail]
// Prints `RESULT <json>`.
import { readFileSync } from 'node:fs';
import { performance } from 'node:perf_hooks';
const arg = (n, f) => process.argv.find((a) => a.startsWith(`--${n}=`))?.slice(n.length + 3) ?? f;
const LAB = arg('lab');
const FILE = arg('requests');
const [id, i] = arg('one').split(':').map(Number);
const WINDOW = Number(arg('window', '85'));
const BUDGET = arg('budget', '');
const DETAIL = process.argv.includes('--detail');
const DSX = JSON.parse(arg('dsx', '{}'));
globalThis.__DSX = DSX;

const { restoreRequest } = await import(`${LAB}/scripts/procgen/seedlingSolveRequestCodec.js`);
const { levelSourceFromAtlas } = await import(`${LAB}/frontend/modules/seedlingDemo/atlasSource.js`);
const { solveAnytime, ANYTIME_PASSES } = await import(`${LAB}/frontend/modules/seedlingDemo/jsRuntimeSolver.js`);

let line = null;
for (const l of readFileSync(FILE, 'utf8').split('\n')) {
    if (!l) continue;
    if (!l.startsWith(`{"id":${id},`)) continue;
    line = JSON.parse(l);
    break;
}
const cap = line.requests.find((r) => r.i === i);
const req = restoreRequest(cap.request);
req.upgradeWindowWork = WINDOW > 0 ? WINDOW : null;
if (BUDGET) req.budgetWork = Number(BUDGET);
const { passes = ANYTIME_PASSES, source, ...tape } = req;
const levelSource = levelSourceFromAtlas(source.records);

// ── the instrument ───────────────────────────────────────────────────────────
let pass = null;
let units = 0;
const sitesByPass = {};
const unitTimes = []; // [pass, site, n, t]
const dashCalls = [];
let cur = null;
const turnsOf = (wps, x0, y0) => {
    let turns = 0; let px = 0; let px0 = x0; let py0 = y0; let dir = null;
    for (const w of wps) {
        const dx = w.x - px0; const dy = w.y - py0;
        px += Math.hypot(dx, dy);
        const d = Math.abs(dx) >= Math.abs(dy) ? (dx >= 0 ? 'E' : 'W') : (dy >= 0 ? 'S' : 'N');
        if (dir !== null && d !== dir && Math.hypot(dx, dy) > 1) turns += 1;
        if (Math.hypot(dx, dy) > 1) dir = d;
        px0 = w.x; py0 = w.y;
    }
    return { turns, px: Math.round(px) };
};
globalThis.__DS = {
    now: () => performance.now(),
    pass(p) { pass = p; },
    unit(site, n) {
        units = n;
        (sitesByPass[pass] ??= {})[site] = ((sitesByPass[pass] ??= {})[site] ?? 0) + 1;
        if (cur) { cur.units += 1; cur.sites[site] = (cur.sites[site] ?? 0) + 1; }
    },
    dashStart(run, wps, opts) {
        const st = run.state ?? null;
        const x0 = st?.x ?? null; const y0 = st?.y ?? null;
        const bodies = (run.entities?.('strikeBodies') ?? []).length;
        const geo = x0 === null ? { turns: null, px: null } : turnsOf(wps, x0, y0);
        cur = { pass, level: run.level ?? null, tick: run.ticksCompleted, u0: units, units: 0, sites: {}, t0: performance.now(),
            wps: wps.length, ...geo, bodies, stand: Boolean(opts.stand), narrow: opts.iframeLapse === false,
            previews: [], pms: 0 };
        return cur;
    },
    base(legs) { if (cur) { cur.baseLegs = legs; cur.baseLongest = Math.max(0, ...legs); } },
    preview(row, ms, walk) {
        if (!cur) return;
        cur.pms += ms;
        cur.previews.push([row.at, row.presses, row.kind ?? (row.certified ? 'certified' : null), row.ticks, Math.round(ms * 100) / 100,
            walk?.samples?.length ?? null]);
    },
    dashEnd(rec, out) {
        rec.ms = performance.now() - rec.t0; delete rec.t0;
        rec.baseline = out?.baseline ?? null; rec.ticks = out?.ticks ?? null; rec.saved = out?.saved ?? null;
        rec.planned = Boolean(out?.plan); rec.windows = out?.windows ?? null;
        rec.why = out?.why ? (out.why === 'deadline' ? 'deadline' : out.why.slice(0, 60)) : null;
        const kinds = {};
        for (const c of out?.candidates ?? []) { const k = c.certified ? 'certified' : c.kind; kinds[k] = (kinds[k] ?? 0) + 1; }
        rec.kinds = kinds; rec.scanned = out?.candidates?.length ?? 0;
        if (!DETAIL) delete rec.previews;
        dashCalls.push(rec);
        cur = null;
    },
};

const t0 = performance.now();
const answer = solveAnytime({ ...tape, source, levelSource }, { passes, onPass: (a) => { if (!a.ok) console.error("PASS", a.pass, String(a.message).slice(0, 600)); } });
const ms = performance.now() - t0;
const rows = (answer.passes ?? []).map((r) => ({ ...r }));
const shipped = answer.ok ? { pass: answer.plan.pass, ticks: answer.plan.solution.length } : null;
console.log(`RESULT ${JSON.stringify({ id, i, window: WINDOW, dsx: DSX, budget: req.budgetWork, ok: answer.ok, kind: answer.kind ?? null,
    work: answer.work ?? null, ms: Math.round(ms), shipped, passes: rows, sitesByPass, dashCalls,
    message: answer.ok ? null : String(answer.message ?? '').slice(0, 200) })}`);

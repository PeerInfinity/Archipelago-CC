// One captured request at its own budget/window (shipped defaults), the stand-tail audit installed (wt's alt patch).
import { readFileSync } from 'node:fs';
const [TREE, FILE, ONE] = process.argv.slice(2);
const [id, i] = ONE.split(':').map(Number);
const M = (p) => import(`${TREE}/frontend/modules/seedlingDemo/${p}`);
const { restoreRequest } = await import(`${TREE}/scripts/procgen/seedlingSolveRequestCodec.js`);
const { levelSourceFromAtlas } = await M('atlasSource.js');
const js = await M('jsRuntimeSolver.js');
const sb = await M('solverBot.js');
let line = null;
for (const l of readFileSync(FILE, 'utf8').split('\n')) { if (l.startsWith(`{"id":${id},`)) { line = JSON.parse(l); break; } }
const { passes = js.ANYTIME_PASSES, source, ...tape } = restoreRequest(line.requests.find((r) => r.i === i).request);
sb.DASH_STAND_WHOLE.audit = [];
const a = js.solveAnytime({ ...tape, source, levelSource: levelSourceFromAtlas(source.records) }, { passes });
console.log(`RESULT ${JSON.stringify({ id, i, key: line.key, ok: a.ok, ticks: a.ok ? a.plan.solution.length : null,
    passes: a.passes.map(({ ms, ...r }) => r), audit: sb.DASH_STAND_WHOLE.audit })}`);

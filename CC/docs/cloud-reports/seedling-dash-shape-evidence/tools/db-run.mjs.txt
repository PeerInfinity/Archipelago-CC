#!/usr/bin/env node
// DB lab runner (scratch). One captured request, one window, this process. Switches come from env (SEEDLING_DASH_*,
// LAB_DASH_WINDOW). node db-run.mjs --lab=<dir> --requests=<jsonl> --one=<id>:<i> --window=<n|0> [--capture]
import { readFileSync } from 'node:fs';
import { performance } from 'node:perf_hooks';
const arg = (n, f) => process.argv.find((a) => a.startsWith(`--${n}=`))?.slice(n.length + 3) ?? f;
const LAB = arg('lab'); const FILE = arg('requests');
const [id, i] = arg('one').split(':').map(Number);
const WINDOW = Number(arg('window', '85'));
const CAPTURE = process.argv.includes('--capture');
const { restoreRequest } = await import(`${LAB}/scripts/procgen/seedlingSolveRequestCodec.js`);
const { levelSourceFromAtlas } = await import(`${LAB}/frontend/modules/seedlingDemo/atlasSource.js`);
const { solveAnytime, ANYTIME_PASSES } = await import(`${LAB}/frontend/modules/seedlingDemo/jsRuntimeSolver.js`);
const sb = await import(`${LAB}/frontend/modules/seedlingDemo/solverBot.js`);
let line = null;
for (const l of readFileSync(FILE, 'utf8').split('\n')) { if (l.startsWith(`{"id":${id},`)) { line = JSON.parse(l); break; } }
const cap = line.requests.find((r) => r.i === i);
const req = restoreRequest(cap.request);
req.upgradeWindowWork = WINDOW > 0 ? WINDOW : null;
const { passes = ANYTIME_PASSES, source, ...tape } = req;
const levelSource = levelSourceFromAtlas(source.records);
const caps = [];
if (CAPTURE) globalThis.__DBCAP = (run, wps, opts) => caps.push({ level: run.level, tick: run.ticksCompleted,
    x: run.state.x, y: run.state.y, wps: wps.map((w) => ({ x: w.x, y: w.y })), opts: { ...opts, stand: opts.stand ?? null } });
const t0 = performance.now();
const answer = solveAnytime({ ...tape, source, levelSource }, { passes });
const ms = performance.now() - t0;
console.log(`RESULT ${JSON.stringify({ id, i, window: WINDOW, budget: req.budgetWork, ok: answer.ok, kind: answer.kind ?? null,
    work: answer.work ?? null, ms: Math.round(ms), shipped: answer.ok ? { pass: answer.plan.pass, ticks: answer.plan.solution.length } : null,
    passes: answer.passes ?? [], switches: { legPre: sb.DASH_LEG_PRECHECK.enabled, noSwing: sb.DASH_NO_SWING_PREFIX.enabled,
        keep: sb.DASH_KEEP_PARTIAL.enabled, labDash: process.env.LAB_DASH_WINDOW ?? null },
    ...(CAPTURE ? { caps, tapeKeys: Object.keys(tape), boot: tape.boot ?? null } : {}) })}`);

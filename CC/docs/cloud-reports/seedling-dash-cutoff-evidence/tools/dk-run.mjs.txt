#!/usr/bin/env node
// DK runner (scratch): df-run.mjs + the chain instrument. One captured request, one window, one fresh process.
import { readFileSync } from 'node:fs';
import { performance } from 'node:perf_hooks';
const arg = (n, f) => process.argv.find((a) => a.startsWith(`--${n}=`))?.slice(n.length + 3) ?? f;
const LAB = arg('lab'); const FILE = arg('requests');
const [id, i] = arg('one').split(':').map(Number);
const WINDOW = Number(arg('window', '85'));
const STAMP = JSON.parse(arg('stamp', '{}'));
const recs = [];
globalThis.__DK = { cur: null, pass: null, done: (rec, out) => {
    const o = out ?? {};
    const cands = o.candidates ?? [];
    const cert = cands.filter((c) => c.certified);
    const L = Math.max(0, ...(rec.baseLegs ?? [0]));
    const li = (rec.baseLegs ?? []).indexOf(L);
    recs.push({ pass: globalThis.__DK.pass, wps: rec.wps, baseline: rec.baseline ?? null, truncated: rec.truncated ?? null,
        baseLegs: rec.baseLegs ?? null, L, li, legStats: rec.legStats ?? null, wpsXY: rec.wpsXY ?? null, start: rec.start ?? null, level: rec.level, startTick: rec.startTick, terrain: rec.terrain ?? null, verdictC: rec.verdictC ?? null,
        dkLong: (rec.dkRuns ?? []).filter((l) => l.n > 195),
        planned: !!o.plan, kept: o.kept ?? null, ticks: o.ticks ?? null, saved: o.saved ?? null, windows: o.windows ?? null,
        finalLegs: o.legs ?? null, scanned: o.scanned ?? cands.length, why: (o.why ?? '').slice(0, 70),
        kinds: cands.reduce((m, c) => { const k = c.certified ? 'CERT' : c.kind; m[k] = (m[k] ?? 0) + 1; return m; }, {}),
        minCandLongest: cands.length ? Math.min(...cands.map((c) => c.longest ?? Infinity)) : null,
        firstCert: cert.length ? { at: cert[0].at, prefix: cert[0].prefix, longest: cert[0].longest, ticks: cert[0].ticks, legs: cert[0].legsAll } : null });
} };
const { restoreRequest } = await import(`${LAB}/scripts/procgen/seedlingSolveRequestCodec.js`);
const { levelSourceFromAtlas } = await import(`${LAB}/frontend/modules/seedlingDemo/atlasSource.js`);
const { solveAnytime, ANYTIME_PASSES } = await import(`${LAB}/frontend/modules/seedlingDemo/jsRuntimeSolver.js`);
const sb = await import(`${LAB}/frontend/modules/seedlingDemo/solverBot.js`);
const dow = await import(`${LAB}/frontend/modules/seedlingDemo/dashOnlyWindowHook.js`);
let line = null;
for (const l of readFileSync(FILE, 'utf8').split('\n')) { if (l.startsWith(`{"id":${id},`)) { line = JSON.parse(l); break; } }
const cap = line.requests.find((r) => r.i === i);
const req = { ...restoreRequest(cap.request), ...STAMP };
const capturedWindow = req.upgradeWindowWork ?? null;
req.upgradeWindowWork = WINDOW > 0 ? WINDOW : null;
if (process.argv.includes('--nobudget')) req.budgetWork = null;
const { passes = ANYTIME_PASSES, source, ...tape } = req;
const levelSource = levelSourceFromAtlas(source.records);
const t0 = performance.now();
const answer = solveAnytime({ ...tape, source, levelSource }, { passes });
const ms = performance.now() - t0;
console.log(`RESULT ${JSON.stringify({ id, i, window: WINDOW, capturedWindow, budget: req.budgetWork, ok: answer.ok, kind: answer.kind ?? null,
    work: answer.work ?? null, ms: Math.round(ms), shipped: answer.ok ? { pass: answer.plan.pass, ticks: answer.plan.solution.length } : null,
    passes: answer.passes ?? [], stamp: STAMP, read: { dashShape: answer.dashShape ?? null, dashOnlyWindow: answer.dashOnlyWindow ?? null },
    defaults: { legPre: sb.DASH_LEG_PRECHECK.enabled, noSwing: sb.DASH_NO_SWING_PREFIX.enabled, keep: sb.DASH_KEEP_PARTIAL.enabled,
        dashOnly: dow.DASH_ONLY_WINDOW.enabled }, corridors: recs })}`);
